					\item Take the notion of
					\begin{gather*}
						C^\infty(X):=\cap C^k(X)

					\item Take the following notion\begin{gather*}
						supp(u):=\overline{\{x|f(x)\not=0\}}

					\item The test functions: Take the notion of
					\begin{gather*}
					C_0^k(X):=\{u\in C^k(X)|supp(u)\mbox{ is compact}\}
					\end{gather*}
				\end{itemize}
			\end{Def}
			
			\subsection{Convolution}
				\begin{Def}
					If $u$ and $v$ are in $C(\RR^n)$ and either one has compact support, then the convolution $u*v$ is the continuous function defined by
					\begin{gather*}
						(u*v)(x):=\int u(x-y)v(y)dy
					\end{gather*}
				\end{Def}
