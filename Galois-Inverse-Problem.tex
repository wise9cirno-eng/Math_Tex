	\begin{OP}
		(\textbf{Inverse Galois Problem})
		
		Whether or not \underline{every} finite group $G$ appears as the Galois group of some Galois extension of the rational number $\mathbb{Q}$. i.e.
		
		\begin{gather*}
			Gal(K/\mathbb{Q})=G
		\end{gather*}
	\end{OP}
	
	
	Equivalently, we should descibe the property of the rational number field
	
	\begin{gather*}
		Gal(\bar{\mathbb{Q}}/\mathbb{Q})
	\end{gather*}
	
	More genrally
	
	\begin{Def}
		Let $G$ be a group, then $G$ is \textbf{relizable} over $K$ if there exists some $L$ such that
		
		\begin{gather*}
			Gal(L/K)=G
		\end{gather*}
	\end{Def}
