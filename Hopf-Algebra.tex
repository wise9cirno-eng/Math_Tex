		\section{(Associative) Algebras, and Coalgebras}
		\subsection{Review:Algebras}
		\begin{Def}
			An (associative) algebra is a triple $(A,m,\eta)$ with:
			\begin{itemize}
				\item $A$ is $k$-vector space;
				\item Operation $m:A\otimes_k A\rightarrow A$; Bilinear.
				\item Operation $\eta:k\rightarrow A$ which is linear.
			\end{itemize}
			
			And we have
			\begin{center}
				\begin{tikzcd}
					A\otimes A\otimes A\arrow[r,"\cong"] &(A\otimes A)\otimes A\arrow[r,"m\otimes id_A"]&A\otimes A\arrow[r,"m"]&A\\
					&&=\\
					A\otimes A\otimes A\arrow[r,"\cong"]&A\otimes (A\otimes A)\arrow[r,"id_A\otimes m"] &A\otimes A\arrow[r,"m"]& A
				\end{tikzcd}
			\end{center}
			
			And the composition
			\begin{center}
				\begin{tikzcd}
					A\cong k\otimes A\arrow[r,"\eta\otimes id_A"]& A\otimes A\arrow[r,"m"]&A\\
					A\cong A\otimes k\arrow[r,"id_A\otimes \eta"]&A\otimes A\arrow[r,"m"]&A
				\end{tikzcd}
			\end{center}
			
			are identity. i.e.
			\begin{gather*}
				m\circ (m\otimes id_A)=m\circ (id_A\otimes m)\\
				m\circ (\eta\otimes id_A)=m\circ (id_A\otimes \eta)
			\end{gather*}
		\end{Def}
		\subsection{Opposite Algebras}
		\begin{Def}
			Let $(A,m,\eta)$ be an algebra over $k$. The algebra
			\begin{gather*}
				(A,m^{op},\eta)
			\end{gather*}
			
			is the \textbf{opposite algebra}.
			
			By defition, the product $m^{op}$ is defined as $a\bullet b:=b\cdot a$.
		\end{Def}
		
		Thus
		\begin{Def}
			An algebra is \textbf{commutative} iff $m^{op}=m$.
		\end{Def}

			\begin{Def}
				Take the following notions: Given algebra $(A,m,\eta)$, then the coalgebras $(C,\Delta, \epsilon)$,
				\begin{center}
					\begin{tikzcd}
						C\arrow[r,"\Delta"]&C\otimes C\arrow[r]\arrow[r,"\Delta \otimes id_C"]&(C\otimes C)\otimes C\simeq C\otimes C\otimes C\\
						C\arrow[r,"\Delta"]&C\otimes C\arrow[r,"id_C\otimes \Delta "]& C\otimes (C\otimes C)\simeq C\otimes C\otimes C
					\end{tikzcd}
				\end{center}
				
				And requiring
				\begin{center}
					\begin{tikzcd}
						C\arrow[r,"\Delta"]&C\otimes C\arrow[r,"\epsilon\otimes id_C"]&k\otimes C\simeq C\\
						C\arrow[r,"\Delta"]&C\otimes C\arrow[r,"Id_C\otimes \epsilon"]&C\otimes k\simeq C
					\end{tikzcd}
				\end{center}
				
				be the identity map $I_C$. i.e.
				\begin{gather*}
					(\Delta\otimes Id_C)\circ \Delta=(Id_C\otimes \Delta)\circ \Delta\\
					(\epsilon\otimes I_C)\circ \Delta=I_C=(I_C\otimes \epsilon)\circ \Delta
				\end{gather*}
			\end{Def}
