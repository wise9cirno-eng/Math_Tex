		\section{Space $\mathcal{D}(\Omega)$}
		
		\begin{Def}
			Take the following notation:
			
			\begin{itemize}
				\item $C_0^\infty(\Omega)$ with a $\RR$-module structure;
				\item $D_k(\Omega):=\{f\in C_0^\infty|supp(f)\subseteq K\}$.
			\end{itemize}
			
			Define the seminorm on $D_K(\Omega)$ by:
			
			\begin{gather*}
				p_{k,m}(f):=\sup_{|s|\leq m,x\in K} |D^s f(x)|,m<\infty
			\end{gather*}
		\end{Def}
