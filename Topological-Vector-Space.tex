		\section{Definition}
			\subsection{Linear Structure, and Compatibilities}
				\begin{Def}
					Let $E$ be a vector space over the field of complex field $F$. Let 
					\begin{gather*}
						A_v:E\times E\rightarrow E\qquad (x,y)\mapsto x+y\\
						M_s:F\times E\rightarrow E\qquad (\lambda,x)\mapsto \lambda x
					\end{gather*}
					Then, a topology $\tau$ on $E$ is said to be \textbf{compatible with linear structure of }$E$ if:
					\begin{itemize}
						\item $A_v:(E\times E,\tau\times \tau)\rightarrow (E,\tau)$ is continuous under product topology;
						\item $M_s:(F\times E,\mathcal{F}\times \tau)\rightarrow (E,\tau)$ is continuous under product topology. 
					\end{itemize}
				\end{Def}
				Then,
				\begin{Def}
					A vector space compactible with linear structure of $E$ is called \textbf{topological vector space}(TVS).
				\end{Def}
				\textbf{Remark} The topology is 'translation invariant'.
