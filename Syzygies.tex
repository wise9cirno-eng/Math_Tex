		\section{Hilbert's Theorem on Syzygies}
		
			\begin{prob}
				Let: $S=\CC[z_0,\dots,z_r]$ in $r+1$ variables over the complex numbers. Let $E$ be a finitely generated graded $S$-module. Choose homogeneous generators:
				
				\begin{gather*}
					m_1,\dots,m_b\in E
				\end{gather*}
				
				with $deg(m_i)=a_{0,i}$. These determine a surjective mapping
				
				\begin{gather*}
					\oplus S(-a_{0,j})\stackrel{\epsilon}{\rightarrow} E\rightarrow 0\qquad (f_1,\dots,f_b)\mapsto \sum f_i\cdot m_i
				\end{gather*}
				
				Then, choose $ker(\epsilon)$ and continue apply the method above, we build a sequence
				
				\begin{center}
					\begin{tikzcd}
						\dots\arrow[r]&P_2=\oplus S(-a_2,j)\arrow[r]&P_1=\oplus S(-a_{1,j})\arrow[r]&E\arrow[r]&0
					\end{tikzcd}
				\end{center}
			\end{prob}
			
			If at each stage we choose the generators minimally in a suitable sense, we can arrange that the matrices defining the $\delta_i$ not have non-zero constant entries. In this case: It is elementary that $P_\bullet$ is unique up to ismomorphism.
			
			\begin{theorem}
				(Hilbert) The process just described terminates after at most $r+1$ steps.
				
				In other words, any \underline{finitely generated} $S$-module $E$ has a minimal graded free resolution of length at most $r+1$, up to isomorphism.
			\end{theorem}
			
			\textbf{Remark} Syzygies and Varieties:
			
			Notice that, the base ring $S$ is Noetherian. Any variety can be represented as an ideal generated by finite numbers of generators,
			
			\begin{gather*}
				X=V(f_1,\dots,f_r)
			\end{gather*}
			
			Then, consider the syzygy of 
			
			\begin{gather*}
				\oplus S(-d) \rightarrow A(X)\rightarrow 0
			\end{gather*}
			
			
			\begin{example}
				(Resolution of the Twisted Quartic Curve) Consider the following curve
				
				\begin{gather*}
					\mathbb{P}^1\hookrightarrow \mathbb{P}^3\qquad [s,t]\mapsto [s^4,s^3t,st^3,t^4]
				\end{gather*}
				
				Take the ideal $I_C\subseteq S=\mathbb{C}[x,y,z,w]$ of $C$, generated by a quadric and three cubics:
				
				\begin{gather*}
					Q=yz-xw,\begin{cases}
						F_1=z^3-yw^2\\
						F_2=xz^2-y^2w\\
						F_3=y^3-x^2z
					\end{cases}
				\end{gather*}
				
				Then, the module of Syzygies among these generators is spanned by the four relations:
				
				\begin{gather*}
					\begin{cases}
						z^2Q-yF_1+wF_2=0\\
						ywQ-xF_1+zF_2=0\\
						xzQ-yF_2-wF_3=0\\
						y^2Q-xF_2-zF_3=0
					\end{cases}
				\end{gather*}
			\end{example}
			
			\begin{prob}
				Is there any structure to the toatlity of all resolutions of some large class of $S$-modules.
			\end{prob}
			
			There are very few modules $E$for which one can write down the resolution $P_\bullet$ explicitly. Instead, much of our attention will be fucused on the question relating to the \textbf{grading} of $P_\bullet$. This involves two sorts of invariants. 
			
			\begin{Def}
				The \textbf{(graded) Betti numbers} $b_{i,j}=b_{i,j}(E)$ are dsefined by writting
				
				\begin{gather*}
					P_i=\oplus_j S(-j)^{b_{i,j}}
				\end{gather*}
				
				with $\beta_{i,j}$ numbers in degree of $j$. 
			\end{Def}
			
			Another definition of Betti numbers is given in algebraic topology. Those definitions coincide in Homological algebra.
			
			\begin{prob}
				Which Betti numbers $b_{ij}(E)$ are nonzero?
			\end{prob}
			
			\begin{Def}
				Define:
				
				\begin{gather*}
					reg(E):=\max\{j-i|b_{i,j\}(E)\not=0}
				\end{gather*}
				
				Then, $reg(E)\leq m\Leftrightarrow E$ is generated in degrees$\leq m$ and for every $i$the $i$th module of syzygies of $E$ is generated in degrees$\leq i+m$.
			\end{Def}
